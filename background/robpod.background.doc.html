def loop(func=None, times=None, delay=1):
    """Execute a function repeatedly with a delay between executions.

    The decorated function is called repeatedly until the specified
    number of executions is reached. If times is None, it runs
    indefinitely.

    This decorator can be used with or without parentheses.

    Args:
        func (callable, optional): Function to decorate. Usually omitted
            when passing decorator options.
        times (int, optional): Number of times to execute the function.
            If None, the function runs indefinitely. Defaults to None.
        delay (int | float, optional): Delay in seconds between
            executions. Defaults to 1.

    Returns:
        callable: A decorated function that repeatedly executes the
        wrapped function. Its return value is not propagated.

    Examples:
        Run a function indefinitely, once per second:

            @loop
            def monitor_robot():
                print(get_state())

        Execute a function ten times with a half-second delay:

            @loop(times=10, delay=0.5)
            def blink_output():
                set_digital_output(0, ON)

    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            count = 0

            while times is None or count < times:
                func(*args, **kwargs)
                count += 1
                time.sleep(delay)

        return wrapper

    if func is not None:
        return decorator(func)

    return decorator

def on(condition, duration=0):
    """Create a rule that triggers once when a condition becomes true.

    The action executes once after the condition has remained true for
    at least the specified duration. The rule will not trigger again
    until the condition becomes false and then true again.

    Args:
        condition (callable): Function returning a truthy or falsy value.
        duration (float, optional): Minimum continuous time in seconds
            that the condition must remain true before the action runs.
            Defaults to 0, meaning the action runs on the first eligible
            evaluation after the condition becomes true.

    Returns:
        RuleWithDo: Builder used to configure the action and start the rule.

    Examples:
        Log once when the robot program enters the running state:

            on(
                lambda: get_program_state() == ProgramStatus.RUNNING
            ).do(
                lambda: log("RUNNING")
            ).start()

        Log once when the program enters the paused state:

            on(
                lambda: get_program_state() == ProgramStatus.PAUSED
            ).do(
                lambda: log("PAUSED")
            ).start()

        Log once when the program enters the stopped state:

            on(
                lambda: get_program_state() == ProgramStatus.STOPPED
            ).do(
                lambda: log("STOPPED")
            ).start()

        Trigger only if the robot remains paused for at least three seconds:

            on(
                lambda: get_program_state() == ProgramStatus.PAUSED,
                duration=3,
            ).do(
                lambda: log("Program has been paused for 3 seconds")
            ).start()

    Note:
        The condition is evaluated periodically, so the action may run
        slightly later than the specified duration due to the polling
        interval and execution time.
    """
    rule = Rule(condition, fire_once=True, duration=duration)
    return RuleWithDo(rule)

def retry(func=None, max_retries=None, delay=1):
    """Retry a function when it raises an exception.

    The decorated function is executed repeatedly until it succeeds or
    the maximum number of retries is reached. A delay is applied between
    failed attempts.

    This decorator can be used with or without parentheses.

    Args:
        func (callable, optional): Function to decorate. Usually omitted
            when passing decorator options.
        max_retries (int, optional): Maximum number of retries after the
            initial attempt. If None, retries indefinitely. Defaults to
            None.
        delay (int | float, optional): Delay in seconds between failed
            attempts. Defaults to 1.

    Returns:
        callable: A decorated function that retries on exceptions and
        returns the wrapped function's result when successful.

    Raises:
        Exception: Re-raises the last exception when the maximum number
            of retries is reached.

    Examples:
        Retry indefinitely until the function succeeds:

            @retry
            def connect():
                return establish_connection()

        Retry up to three times after the initial attempt:

            @retry(max_retries=3, delay=2)
            def read_sensor():
                return sensor.read()
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            attempt = 0

            while max_retries is None or attempt <= max_retries:
                try:
                    return func(*args, **kwargs)
                except Exception:
                    if max_retries is not None and attempt >= max_retries:
                        raise

                    attempt += 1
                    time.sleep(delay)

        return wrapper

    if func is not None:
        return decorator(func)

    return decorator

def when(condition):
    """Create a rule that executes its action while a condition is true.

    The condition is evaluated approximately every 0.5 seconds. Whenever
    the condition evaluates to true, the action is executed. Therefore,
    the action may run repeatedly for as long as the condition remains true.

    Args:
        condition (callable): Function returning a truthy or falsy value.

    Returns:
        RuleWithDo: Builder used to configure the action and start the rule.

    Examples:
        Log repeatedly while the robot program is running:

            when(
                lambda: get_program_state() == ProgramStatus.RUNNING
            ).do(
                lambda: log("Program is running")
            ).start()

        Turn on an output while the program is running:

            when(
                lambda: get_program_state() == ProgramStatus.RUNNING
            ).do(
                lambda: set_digital_output(0, ON)
            ).start()

    Note:
        The action is not edge-triggered. If the condition stays true,
        it is executed on every evaluation, approximately twice per second
        with the default delay.
    """
    rule = Rule(condition, fire_once=False)
    return RuleWithDo(rule)

class DoosanRobotClient(_RobotClient):

    def __init__(self, host):
        _RobotClient.__init__(self, host)  # call parent constructor
        self.modbus_client = modbus.ModbusClient(host)

    def get_digital_input(self, index):
        return self.modbus_client.read_coils(1, -1 + index, 1)[0]

    def get_configurable_digital_input(self, index):
        return get_digital_input(index)

    def get_digital_output(self, index):
        return self.modbus_client.read_coils(1, 19 + index, 1)[0]

    def get_configurable_digital_output(self, index):
        return get_digital_output(index)

    def set_configurable_digital_output(self, index, value):
        return set_digital_output(index, value)

    def set_digital_output(self, index, value):
        return self.modbus_client.write_coils(1, 19 + index, [value])

    def get_tool_digital_output(self, index):
        return self.modbus_client.read_coils(1, 41 + index, 1)[0]

    def get_tool_digital_input(self, index):
        return self.modbus_client.read_coils(1, 35 + index, 1)[0]

    def set_tool_digital_output(self, index, value):
        return self.modbus_client.write_coils(1, 41 + index, [value])

    def get_joint_pose(self):
        values = self.modbus_client.read_holding_registers(1, 270, 6)

        result = []
        for val in values:
            result.append((modbus.to_signed_value(val)/ 10))

        return result

    def get_joint_velocity(self):
        values = self.modbus_client.read_holding_registers(1, 280, 6)

        result = []
        for val in values:
            result.append((modbus.to_signed_value(val) / 10))

        return result

    def get_tcp_pose(self):
        values = self.modbus_client.read_holding_registers(1, 400, 6)

        result = []
        for val in values:
            result.append((modbus.to_signed_value(val) / 10))

        return result

    def get_tcp_velocity(self):
        values = self.modbus_client.read_holding_registers(1, 410, 6)

        result = []
        for val in values:
            result.append((modbus.to_signed_value(val) / 10))

        return result

    def get_tcp_offset(self):
        values = self.modbus_client.read_holding_registers(1, 420, 6)

        result = []
        for val in values:
            result.append((modbus.to_signed_value(val) / 10))

        return result

    def get_register(self, index):
        return self.modbus_client.read_holding_registers(1, index, 1)[0]

    def set_register(self, index, value):
        return self.modbus_client.write_register(1, index, value)

    def get_state(self):
        status = self.modbus_client.read_holding_registers(1, 259, 1)[0]

        if status == 0 or status == 1 or status == 2:
            return RobotStatus.BACKDRIVE
        elif status == 3:
            return RobotStatus.SERVO_OFF
        elif status == 4:
            return RobotStatus.INITIALIZING
        elif status == 5:
            return RobotStatus.SAFETY_STOP
        elif status == 6:
            return RobotStatus.EMERGENCY_STOP
        elif status == 7 or status == 8 or status == 9 or status == 10:
            return RobotStatus.RECOVERY
        elif status == 11 or status == 12:
            return RobotStatus.SERVO_ON
        elif status == 13:
            return RobotStatus.FREE_DRIVE
        elif status == 14 or status == 15 or status == 16:
            return RobotStatus.SERVO_ON
        elif status == 17 or status == 18:
            return RobotStatus.SERVO_ON_REDUCED
        elif status == 19 or status == 20:
            return RobotStatus.FREE_DRIVE
        else:
            return RobotStatus.SERVO_OFF

class Rule:
    def __init__(self, condition, fire_once=False, duration=0):
        self.condition = condition
        self.action = None
        self.delay = 0.5
        self.fire_once = fire_once
        self.duration = duration
        self.condition_started_at = None
        self.prev_condition = False

    def _run(self):
        while True:
            try:
                condition_value = self.condition()

                if self.fire_once:
                    if condition_value:
                        # OFF -> ON: start the timer
                        if not self.prev_condition:
                            self.condition_started_at = time.time()

                        # Condition has remained ON for the required duration
                        if (
                            self.condition_started_at is not None
                            and time.time() - self.condition_started_at >= self.duration
                        ):
                            self.action()

                            # Prevent firing again while condition stays ON
                            self.condition_started_at = None
                    else:
                        # ON -> OFF: reset the timer
                        self.condition_started_at = None

                else:
                    if condition_value:
                        self.action()

                self.prev_condition = condition_value
                time.sleep(self.delay)

            except Exception as e:
                time.sleep(self.delay)
                print(e)

    def _start(self):
        thread = threading.Thread(target=self._run)
        thread.daemon = False
        thread.start()

def get_state(self):
    safety_mode = self.modbus_client.read_holding_registers(1, 266, 1)[0]
    status = self.modbus_client.read_holding_registers(1, 258, 1)[0]

    if safety_mode == 1 or safety_mode == 2:
        if status == 3:
            return RobotStatus.SERVO_OFF
        elif status == 4 or status == 5:
            return RobotStatus.IDLE
        elif status == 6:
            return RobotStatus.BACKDRIVE
        elif status == 7:
            if safety_mode == 2:
                return RobotStatus.SERVO_ON_REDUCED
            else:
                return RobotStatus.SERVO_ON
        else:
            return RobotStatus.SERVO_OFF
    elif safety_mode == 6 or safety_mode == 7:
        return RobotStatus.EMERGENCY_STOP
    elif safety_mode == 4:
        return RobotStatus.EMERGENCY_STOP
    else:
        return RobotStatus.SAFETY_STOP

def wait_boot_completed(self, timeout=60):
    deadline = time.time() + timeout

    while True:
        try:
            self.get_program_name()
            return True
        except :
            remaining = deadline - time.time()
            if remaining <= 0:
                raise TimeoutError("Robot did not become available within " + str(timeout) + " seconds")

            time.sleep(2)
